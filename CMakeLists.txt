cmake_minimum_required(VERSION 3.16)
project(miku_fm VERSION 0.5 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(miku_core STATIC firmware/sequencer/sequencer.c firmware/audio/vocal/vocal.c firmware/audio/fvoice/fvoice.c firmware/audio/band/band.c firmware/platform/fm1_runtime.c firmware/ui/stage.c shared/song_io.c)
target_include_directories(miku_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
  target_compile_options(miku_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(miku_demo tools/demo.c)
target_link_libraries(miku_demo PRIVATE miku_core m)
enable_testing()
add_executable(miku_tests tests/test_core.c)
target_link_libraries(miku_tests PRIVATE miku_core m)
add_test(NAME core COMMAND miku_tests)
add_executable(miku_song_io_tests tests/test_song_io.c)
target_link_libraries(miku_song_io_tests PRIVATE miku_core m)
add_test(NAME song_io COMMAND miku_song_io_tests)
add_executable(miku_band_tests tests/test_band.c)
target_link_libraries(miku_band_tests PRIVATE miku_core m)
add_test(NAME band COMMAND miku_band_tests)
add_executable(miku_runtime_tests tests/test_runtime.c)
target_link_libraries(miku_runtime_tests PRIVATE miku_core m)
add_test(NAME runtime COMMAND miku_runtime_tests)
add_executable(miku_stage_tests tests/test_stage.c)
target_link_libraries(miku_stage_tests PRIVATE miku_core m)
add_test(NAME stage COMMAND miku_stage_tests)
add_test(NAME demo_wav COMMAND miku_demo ${CMAKE_CURRENT_BINARY_DIR}/test.wav ${CMAKE_CURRENT_BINARY_DIR}/test.mik)
add_executable(miku_fvoice_tests tests/test_fvoice.c)
target_link_libraries(miku_fvoice_tests PRIVATE miku_core m)
add_test(NAME fvoice COMMAND miku_fvoice_tests)
add_executable(miku_mora_voice_tests tests/test_mora_voice.c firmware/song/mora.c)
target_link_libraries(miku_mora_voice_tests PRIVATE miku_core m)
add_test(NAME mora_voice COMMAND miku_mora_voice_tests)
add_executable(fvoice_demo tools/fvoice_demo.c)
target_link_libraries(fvoice_demo PRIVATE miku_core m)
# Step 2 modules, each with its own CMakeLists.txt (added when present)
foreach(d firmware/song firmware/audio/fmband firmware/app firmware/audio/pedal)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${d}/CMakeLists.txt)
    add_subdirectory(${d})
  endif()
endforeach()
# Execute the native SELECT branch with the host engine; no substitute UI implementation.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NOT MSVC)
  add_test(NAME live_select COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_live_select.py
    ${CMAKE_SOURCE_DIR} ${CMAKE_BINARY_DIR} ${CMAKE_C_COMPILER})
endif()
if(NOT MSVC)
  add_test(NAME sprite_decode COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_sprite_decode.py
    ${CMAKE_SOURCE_DIR} ${CMAKE_C_COMPILER})
endif()

add_library(miku_performance STATIC firmware/ui/performance.c)
target_include_directories(miku_performance PUBLIC ${CMAKE_SOURCE_DIR})
add_executable(performance_tests tests/test_performance.c)
target_link_libraries(performance_tests PRIVATE miku_performance)
add_test(NAME performance COMMAND performance_tests)
add_executable(performance_preview tools/performance_preview.c)
target_link_libraries(performance_preview PRIVATE miku_performance)
if(NOT MSVC)
  target_compile_options(miku_performance PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(performance_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(performance_preview PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(miku_performance_wire STATIC firmware/app/performance_wire.c)
target_include_directories(miku_performance_wire PUBLIC ${CMAKE_SOURCE_DIR})
add_executable(performance_wire_tests tests/test_performance_wire.c)
target_link_libraries(performance_wire_tests PRIVATE miku_performance_wire)
add_test(NAME performance_wire COMMAND performance_wire_tests)
add_executable(performance_export tools/performance_export.c)
target_link_libraries(performance_export PRIVATE miku_engine miku_performance miku_performance_wire m)
if(NOT MSVC)
  target_compile_options(miku_performance_wire PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(performance_wire_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(performance_export PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME resource_budget COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_resource_budget.py)

add_library(miku_sprite_codec STATIC firmware/ui/sprite_codec.c)
target_include_directories(miku_sprite_codec PUBLIC ${CMAKE_SOURCE_DIR})
add_executable(sprite_codec_tests tests/test_sprite_codec.c)
target_link_libraries(sprite_codec_tests PRIVATE miku_sprite_codec)
add_test(NAME sprite_codec COMMAND sprite_codec_tests)
add_test(NAME sprite_generator COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_SOURCE_DIR}/tests/test_sprite_codec.py)
add_subdirectory(firmware/usb/video)
