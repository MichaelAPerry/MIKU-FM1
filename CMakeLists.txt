cmake_minimum_required(VERSION 3.16)
project(miku_fm VERSION 0.5 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(miku_core STATIC firmware/sequencer/sequencer.c firmware/audio/vocal/vocal.c firmware/audio/fvoice/fvoice.c firmware/audio/band/band.c firmware/platform/fm1_runtime.c firmware/ui/stage.c shared/song_io.c)
target_include_directories(miku_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(NOT MSVC)
  target_compile_options(miku_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(miku_demo tools/demo.c)
target_link_libraries(miku_demo PRIVATE miku_core m)
enable_testing()
add_executable(miku_tests tests/test_core.c)
target_link_libraries(miku_tests PRIVATE miku_core m)
add_test(NAME core COMMAND miku_tests)
add_executable(miku_song_io_tests tests/test_song_io.c)
target_link_libraries(miku_song_io_tests PRIVATE miku_core m)
add_test(NAME song_io COMMAND miku_song_io_tests)
add_executable(miku_band_tests tests/test_band.c)
target_link_libraries(miku_band_tests PRIVATE miku_core m)
add_test(NAME band COMMAND miku_band_tests)
add_executable(miku_runtime_tests tests/test_runtime.c)
target_link_libraries(miku_runtime_tests PRIVATE miku_core m)
add_test(NAME runtime COMMAND miku_runtime_tests)
add_executable(miku_stage_tests tests/test_stage.c)
target_link_libraries(miku_stage_tests PRIVATE miku_core m)
add_test(NAME stage COMMAND miku_stage_tests)
add_test(NAME demo_wav COMMAND miku_demo ${CMAKE_CURRENT_BINARY_DIR}/test.wav ${CMAKE_CURRENT_BINARY_DIR}/test.mik)
add_executable(miku_fvoice_tests tests/test_fvoice.c)
target_link_libraries(miku_fvoice_tests PRIVATE miku_core m)
add_test(NAME fvoice COMMAND miku_fvoice_tests)
add_executable(fvoice_demo tools/fvoice_demo.c)
target_link_libraries(fvoice_demo PRIVATE miku_core m)
