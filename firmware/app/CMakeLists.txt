# Step 2c: the song engine (sequencer events -> voice + band), shared by the device app and the host render tool.
add_library(miku_engine STATIC engine.c ${CMAKE_SOURCE_DIR}/firmware/audio/vfx/vfx.c)
target_link_libraries(miku_engine PUBLIC miku_song fmband miku_core)
target_include_directories(miku_engine PUBLIC ${CMAKE_SOURCE_DIR})
if(NOT MSVC)
  target_compile_options(miku_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(song_render ${CMAKE_SOURCE_DIR}/tools/song_render.c)
target_link_libraries(song_render PRIVATE miku_engine m)
add_executable(mora_render ${CMAKE_SOURCE_DIR}/tools/mora_render.c)
target_link_libraries(mora_render PRIVATE miku_engine m)
add_executable(engine_tests ${CMAKE_SOURCE_DIR}/tests/test_engine.c)
target_link_libraries(engine_tests PRIVATE miku_engine m)
if(NOT MSVC)
  target_compile_options(engine_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(song_render PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mora_render PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME engine COMMAND engine_tests)

add_executable(gen_next_render ${CMAKE_SOURCE_DIR}/tools/gen_next_render.c)
target_link_libraries(gen_next_render PRIVATE miku_engine m)
if(NOT MSVC)
  target_compile_options(gen_next_render PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(lyric_engine_tests ${CMAKE_SOURCE_DIR}/tests/test_lyric_engine.c)
target_link_libraries(lyric_engine_tests PRIVATE miku_engine m)
if(NOT MSVC)
  target_compile_options(lyric_engine_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME lyric_engine COMMAND lyric_engine_tests)
add_executable(factory_mix_tests ${CMAKE_SOURCE_DIR}/tests/test_factory_mix.c)
target_link_libraries(factory_mix_tests PRIVATE miku_engine m)
add_test(NAME factory_mix COMMAND factory_mix_tests)
add_executable(dx7_bank_tests ${CMAKE_SOURCE_DIR}/tests/test_dx7_bank.c)
target_link_libraries(dx7_bank_tests PRIVATE miku_engine)
add_test(NAME dx7_bank COMMAND dx7_bank_tests)
add_executable(dx7_examples ${CMAKE_SOURCE_DIR}/tools/dx7_examples.c)
target_link_libraries(dx7_examples PRIVATE miku_engine)
