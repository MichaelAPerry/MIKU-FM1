add_library(miku_pedal STATIC pedal.c)
target_include_directories(miku_pedal PUBLIC ${CMAKE_SOURCE_DIR})
add_executable(pedal_tests tests/test_pedal.c)
target_link_libraries(pedal_tests PRIVATE miku_pedal)
if(NOT MSVC)
  target_compile_options(miku_pedal PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(pedal_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME pedal COMMAND pedal_tests)
add_executable(pedal_render ${CMAKE_SOURCE_DIR}/tools/pedal_render.c)
target_link_libraries(pedal_render PRIVATE miku_pedal)
if(NOT MSVC)
  target_compile_options(pedal_render PRIVATE -Wall -Wextra -Wpedantic)
endif()
